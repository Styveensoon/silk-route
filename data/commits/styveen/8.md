## Codigo


**Archivo:** `tests/Feature/ServiceTest.php`

```php
<?php

namespace Tests\Feature;

use App\Models\Category;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class ServiceTest extends TestCase
{
    use RefreshDatabase;

    public function test_services_index_page_loads(): void
    {
        $response = $this->get('/services');

        $response->assertStatus(200);
    }

    public function test_a_service_can_be_published(): void
    {
        User::factory()->create();
        $category = Category::create(['name' => 'Tutorias', 'slug' => 'tutorias']);

        $response = $this->post('/services', [
            'title' => 'Clases de calculo diferencial',
            'description' => 'Asesoria personalizada para el examen final.',
            'price' => 250,
            'category_id' => $category->id,
        ]);

        $response->assertRedirect(route('services.index'));

        $this->assertDatabaseHas('services', [
            'title' => 'Clases de calculo diferencial',
            'category_id' => $category->id,
        ]);
    }
}
```
